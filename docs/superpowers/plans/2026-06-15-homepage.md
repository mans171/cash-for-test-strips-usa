> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the full homepage for Cash4TestStripsUSA — navy/green SEO-optimized page with nav, hero (search + geolocation), how-it-works, featured buyers from Supabase, state strip, blog teasers, and footer.

**Architecture:** `app/page.tsx` is a server component that fetches featured buyers from Supabase and composes static section components. The hero search bar and geolocation button live in `components/HeroSearch.tsx`, the only client component. All other components are server-only.

**Tech Stack:** Next.js 16 App Router, TypeScript, Tailwind CSS v4, Supabase JS (anon key, public read)

```tsx
// components/HeroSearch.tsx
'use client'

import { useRouter } from 'next/navigation'
import { useState } from 'react'
import { STATE_NAME_TO_SLUG } from '@/lib/states'

export default function HeroSearch() {
  const router = useRouter()
  const [query, setQuery] = useState('')
  const [geoLoading, setGeoLoading] = useState(false)

  function handleSearch(e: React.FormEvent) {
    e.preventDefault()
    if (query.trim()) {
      router.push(`/directory?q=${encodeURIComponent(query.trim())}`)
    }
  }

  function handleGeolocation() {
    if (!navigator.geolocation) {
      router.push('/directory')
      return
    }
    setGeoLoading(true)
    navigator.geolocation.getCurrentPosition(
      async (pos) => {
        try {
          const { latitude, longitude } = pos.coords
          const res = await fetch(
            `https://api.bigdatacloud.net/data/reverse-geocode-client?latitude=${latitude}&longitude=${longitude}&localityLanguage=en`
          )
          const data = (await res.json()) as { principalSubdivision?: string }
          const slug = STATE_NAME_TO_SLUG[data.principalSubdivision ?? '']
          router.push(slug ? `/sell-test-strips/${slug}` : '/directory')
        } catch {
          router.push('/directory')
        } finally {
          setGeoLoading(false)
        }
      },
      () => {
        setGeoLoading(false)
        router.push('/directory')
      }
    )
  }

  return (
    <div className="max-w-xl mx-auto">
      <form onSubmit={handleSearch} className="flex bg-white rounded-lg overflow-hidden shadow-xl">
        <input
          type="text"
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          placeholder="Search by state, city, or buyer name…"
          className="flex-1 px-5 py-4 text-sm text-gray-800 outline-none"
        />
        <button
          type="submit"
          className="bg-[#27ae60] text-white text-sm font-bold px-6 py-4 hover:bg-[#229954] transition-colors"
        >
          Search →
        </button>
      </form>
      <div className="mt-4 text-center">
        <span className="block text-white/40 text-xs mb-3">or</span>
        <button
          onClick={handleGeolocation}
          disabled={geoLoading}
          className="inline-flex items-center gap-2 bg-white/10 border border-white/25 text-white text-sm font-semibold px-5 py-2.5 rounded-full hover:bg-white/20 transition-colors disabled:opacity-60 disabled:cursor-not-allowed"
        >
          {geoLoading ? '📍 Detecting location…' : '📍 Use My Location'}
        </button>
      </div>
    </div>
  )
}
```

```tsx
// components/FeaturedBuyers.tsx
import { supabase } from '@/lib/supabase'
import BuyerCard from '@/components/BuyerCard'
import Link from 'next/link'
import type { Company } from '@/lib/types'

export default async function FeaturedBuyers() {
  const { data, error } = await supabase
    .from('companies')
    .select('*')
    .eq('active', true)
    .order('featured', { ascending: false })
    .limit(6)

  if (error) console.error('FeaturedBuyers fetch error:', error.message)

  const companies = (data ?? []) as Company[]

  return (
    <section className="py-14 px-8 bg-[#f9fbff]">
      <div className="max-w-5xl mx-auto">
        <p className="text-[11px] font-bold tracking-widest uppercase text-[#27ae60] mb-2">Top Picks</p>

        <div className="grid grid-cols-1 sm:grid-cols-2 md:grid-cols-3 gap-4">
          {companies.map((company) => (
            <BuyerCard key={company.id} company={company} />
```
